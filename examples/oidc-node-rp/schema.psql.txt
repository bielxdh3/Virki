-- Apply ONLY to an independently provisioned RP-owned PostgreSQL database via psql -f.
-- Flow state hashes are opaque, and the transaction fields are AES-256-GCM encrypted.
CREATE TABLE IF NOT EXISTS "VirkiOidcLoginFlow" (
  "stateHash" CHAR(64) PRIMARY KEY,
  "ciphertext" TEXT NOT NULL,
  "expiresAt" TIMESTAMPTZ NOT NULL
);
CREATE INDEX IF NOT EXISTS "VirkiOidcLoginFlow_expiresAt_idx"
  ON "VirkiOidcLoginFlow" ("expiresAt");

-- RP sessions are independent of portal/issuer sessions. Never expose these rows to a browser.
CREATE TABLE IF NOT EXISTS "VirkiRpSession" (
  "idHash" CHAR(64) PRIMARY KEY,
  "issuer" TEXT NOT NULL,
  "subject" TEXT NOT NULL,
  "tokenCiphertext" TEXT NOT NULL,
  "expiresAt" TIMESTAMPTZ NOT NULL
);
CREATE INDEX IF NOT EXISTS "VirkiRpSession_expiresAt_idx"
  ON "VirkiRpSession" ("expiresAt");
